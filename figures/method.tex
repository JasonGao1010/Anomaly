\newcommand{\gridlines}[2]{%
  \begin{scope}
    \clip (#1.south west) rectangle (#1.north east);
    \foreach \t in {0.2,0.4,0.6,0.8}
      \draw[#2!60,line width=.15pt]
        ($(#1.south west)!\t!(#1.north west)$) -- ($(#1.south east)!\t!(#1.north east)$);
    \foreach \t in {0.25,0.5,0.75}
      \draw[#2!60,line width=.15pt]
        ($(#1.south west)!\t!(#1.south east)$) -- ($(#1.north west)!\t!(#1.north east)$);
  \end{scope}
}
\newcommand{\gridnode}[6]{%
  \pgfmathsetmacro{\hh}{2.5+.29*ln(#4)/ln(2)}
  \node[draw=#5!80!black,fill=#5!16,minimum width=2.2mm,
    minimum height=\hh mm,inner sep=0,line width=.3pt] (#1) at (#2,#3) {};
  \gridlines{#1}{#5}
  \node[dim,fill=#5!8,inner sep=.12mm] at (#1.center) {#6};
}
\newcommand{\module}[6]{%
  \node[draw=#5!80!black,fill=#5!12,minimum width=#4 mm,
    minimum height=6mm,inner sep=0,line width=.3pt] (#1) at (#2,#3) {};
  \gridlines{#1}{#5}
  \node[note,align=center,fill=#5!7,inner sep=.3mm] at (#1.center) {#6};
  \learn{#1}
}

  \coordinate (#1-low) at #2; \coordinate (#1-high) at #3;
  \begin{scope}[on background layer]
    \node[region,fit=(#1-low)(#1-high)] (#1) {};
  \end{scope}
  \node[regiontitle,anchor=east,fill=white,inner sep=.2mm]
    at ([xshift=-.6mm]#1.north east) {#4};
}
